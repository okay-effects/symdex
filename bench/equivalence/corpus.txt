=== Escape
--- before
object Escape:
  def quote(s: String): String = "\"" + s.flatMap {
    case '"' => "\\\""
    case '\\' => "\\\\"
    case c => c.toString
  } + "\""
--- after
object Escape:
  def quote(s: String): String = "\"" + s.replace("\\", "\\\\").replace("\"", "\\\"") + "\""

=== Settings
--- before
object Settings:
  def timeout(raw: Option[String]): Either[String, Int] =
    raw.toRight("missing").flatMap(s => s.toIntOption.toRight(s"bad: $s")).filterOrElse(_ > 0, "not positive")
--- after
object Settings:
  def timeout(raw: Option[String]): Either[String, Int] = raw match
    case None => Left("missing")
    case Some(s) => s.toIntOption match
      case Some(n) if n > 0 => Right(n)
      case Some(_) => Left("not positive")
      case None => Left(s"bad: $s")

=== Versions
--- before
object Versions:
  def compare(a: List[Int], b: List[Int]): Int = (a, b) match
    case (Nil, Nil) => 0
    case (Nil, _) => -1
    case (_, Nil) => 1

    a.zipAll(b, 0, 0).find(_ != _).map((x, y) => x.compare(y)).getOrElse(0)
